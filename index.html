// ════════════════════════════════════════════════════════════════ // MTJ Welding CRM — Google Apps Script Backend // Paste this entire file into your Apps Script project, // then create a new deployment (Deploy → New deployment). // ════════════════════════════════════════════════════════════════ var SHEET_NAME = "Leads"; var COLUMNS = [ "id", "name", "datein", "phone", "email", "source", "address", "type", "status", "scope", "notes", "value", "datequoted", "followup", "deposit", "depositdate" ]; // ── Response helpers ────────────────────────────────────────── function makeResponse(payload) { var output = ContentService.createTextOutput(JSON.stringify(payload)); output.setMimeType(ContentService.MimeType.JSON); return output; } function ok(data) { return makeResponse({ ok: true, data: data }); } function err(msg) { return makeResponse({ ok: false, error: msg }); } // ── Sheet helpers ───────────────────────────────────────────── function getSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(COLUMNS); sheet.getRange(1, 1, 1, COLUMNS.length).setFontWeight("bold"); sheet.setFrozenRows(1); } return sheet; } function rowToObj(row) { var obj = {}; for (var i = 0; i < COLUMNS.length; i++) { obj[COLUMNS[i]] = row[i] !== undefined ? String(row[i]) : ""; } return obj; } function objToRow(obj) { return COLUMNS.map(function(col) { return obj[col] !== undefined ? obj[col] : ""; }); } // ── GET — list all leads ────────────────────────────────────── function doGet(e) { try { var action = (e.parameter && e.parameter.action) ? e.parameter.action : "list"; if (action === "list") { var sheet = getSheet(); var data = sheet.getDataRange().getValues(); if (data.length <= 1) { return ok([]); } var leads = []; for (var i = 1; i < data.length; i++) { leads.push(rowToObj(data[i])); } return ok(leads); } return err("Unknown action: " + action); } catch(ex) { return err(ex.toString()); } } // ── POST — upsert / delete / seed ──────────────────────────── function doPost(e) { try { var body = JSON.parse(e.postData.contents); var action = body.action; var sheet = getSheet(); if (action === "upsert") { var lead = body.lead; if (!lead || !lead.id) { return err("Missing lead or id"); } var data = sheet.getDataRange().getValues(); var found = false; for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === String(lead.id)) { sheet.getRange(i + 1, 1, 1, COLUMNS.length).setValues([objToRow(lead)]); found = true; break; } } if (!found) { sheet.appendRow(objToRow(lead)); } return ok({ upserted: lead.id }); } if (action === "delete") { var id = String(body.id); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === id) { sheet.deleteRow(i + 1); return ok({ deleted: id }); } } return err("Lead not found: " + id); } if (action === "seed") { var leads = body.leads; if (!Array.isArray(leads)) { return err("leads must be an array"); } for (var i = 0; i < leads.length; i++) { sheet.appendRow(objToRow(leads[i])); } return ok({ seeded: leads.length }); } return err("Unknown action: " + action); } catch(ex) { return err(ex.toString()); } }
